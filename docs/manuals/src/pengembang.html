§8Runbook operasi

  
Semua perintah dijalankan dari salinan live /opt/lendatagovernance/deploy (project compose openmetadata). Ubah di repo, lalu salin ke salinan live.


# kesehatan
sudo docker compose -p openmetadata ps ; scripts/verify.sh
curl -s http://127.0.0.1/_gateway/health
# mulai / henti / log
sudo docker compose -p openmetadata up -d ; sudo docker compose -p openmetadata stop
sudo docker compose -p openmetadata logs -f --tail=200 openmetadata-server
# terapkan perubahan gateway (nginx.conf, tema, mount panel)
sudo docker compose -p openmetadata up -d gateway ; sudo docker compose -p openmetadata exec gateway nginx -t
# backup (timer harian 01:30 → /srv/lendatagovernance/backup) dan pemulihan
sudo scripts/backup.sh
sudo docker compose -p openmetadata stop openmetadata-server
sudo docker compose -p openmetadata exec -T postgresql pg_restore -U postgres -d openmetadata_db --clean --if-exists \
  < /srv/lendatagovernance/backup/openmetadata_db-<waktu>.dump      # butuh openmetadata.env + jwt/ yang sama
sudo docker compose -p openmetadata start openmetadata-server   # lalu Settings › Applications › Search Indexing › Run
# token loader: UI › Settings › Bots/Users › admin › Access Token (60–90 hari) → OM_TOKEN di secrets/loader.env
scripts/reload-workbook.sh "templates/<workbook>.xlsx"       # muat ulang + selisih cacat
# akun panel /unduh/
scripts/panel-account.sh <user> ; scripts/panel-account.sh --remove <user>
# fase 2: IDS (butuh client IDS + DNS dev-data.len.co.id)
scripts/enable-ids.sh
